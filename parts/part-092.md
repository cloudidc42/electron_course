```rust
// native/src/lib.rs
#![deny(clippy::all)]

use napi_derive::napi;
use napi::{JsBuffer, Result};
use std::time::Instant;

// Simple sync function
#[napi]
pub fn add(a: i64, b: i64) -> i64 {
  a + b
}

// CPU-intensive: hash large data
#[napi]
pub fn sha256_hash(data: Buffer) -> String {
  use sha2::{Sha256, Digest};
  let hash = Sha256::digest(&*data);
  hex::encode(hash)
}

// Parallel processing ด้วย Rayon
#[napi]
pub fn parallel_process(numbers: Vec<f64>) -> Vec<f64> {
  use rayon::prelude::*;
  numbers.par_iter()
    .map(|x| x.sqrt())
    .collect()
}

// Struct ที่ expose ให้ JavaScript
#[napi]
pub struct ImageProcessor {
  width: u32,
  height: u32,
  data: Vec<u8>,
}

#[napi]
impl ImageProcessor {
  #[napi(constructor)]
  pub fn new(width: u32, height: u32) -> Self {
    ImageProcessor {
      width,
      height,
      data: vec![0u8; (width * height * 4) as usize],
    }
  }

  #[napi]
  pub fn width(&self) -> u32 {
    self.width
  }

  #[napi]
  pub fn height(&self) -> u32 {
    self.height
  }

  #[napi]
  pub fn set_pixel(&mut self, x: u32, y: u32, r: u8, g: u8, b: u8, a: u8) {
    if x < self.width && y < self.height {
      let idx = ((y * self.width + x) * 4) as usize;
      self.data[idx] = r;
      self.data[idx + 1] = g;
      self.data[idx + 2] = b;
      self.data[idx + 3] = a;
    }
  }

  #[napi]
  pub fn get_data(&self) -> Vec<u8> {
    self.data.clone()
  }

  // Grayscale filter
  #[napi]
  pub fn grayscale(&mut self) {
    for i in (0..self.data.len()).step_by(4) {
      let r = self.data[i] as f32;
      let g = self.data[i + 1] as f32;
      let b = self.data[i + 2] as f32;
      let gray = (0.299 * r + 0.587 * g + 0.114 * b) as u8;
      self.data[i] = gray;
      self.data[i + 1] = gray;
      self.data[i + 2] = gray;
    }
  }
}

// Async function ด้วย Tokio
#[napi]
pub async fn fetch_and_process(url: String) -> Result<String> {
  // Tokio async runtime
  let client = reqwest::Client::new();
  let body = client.get(&url)
    .send()
    .await
    .map_err(|e| napi::Error::from_reason(e.to_string()))?
    .text()
    .await
    .map_err(|e| napi::Error::from_reason(e.to_string()))?;

  Ok(format!("Downloaded {} bytes", body.len()))
}

// Benchmark function
#[napi]
pub fn benchmark_fib(n: u32) -> u64 {
  let start = Instant::now();
  let result = fib(n);
  let _elapsed = start.elapsed();
  result
}

fn fib(n: u32) -> u64 {
  match n {
    0 => 0,
    1 => 1,
    _ => {
      let mut a: u64 = 0;
      let mut b: u64 = 1;
      for _ in 2..=n {
        let c = a.saturating_add(b);
        a = b;
        b = c;
      }
      b
    }
  }
}
```

## 4. การใช้งานใน Electron

```typescript
// src/main/nativeModule.ts
import { ipcMain } from 'electron'

// โหลด native module
let native: typeof import('@myapp/native') | null = null

function getNative() {
  if (!native) {
    try {
      native = require('@myapp/native')
    } catch (error) {
      console.error('Failed to load native module:', error)
    }
  }
  return native
}

export function registerNativeHandlers(): void {
  ipcMain.handle('native:sha256', async (_, { data }: { data: number[] }) => {
    const mod = getNative()
    if (!mod) return null
    const buffer = Buffer.from(data)
    return mod.sha256Hash(buffer)
  })

  ipcMain.handle('native:parallel-process', async (_, { numbers }: { numbers: number[] }) => {
    const mod = getNative()
    if (!mod) return numbers
    return mod.parallelProcess(numbers)
  })

  ipcMain.handle('native:benchmark', async (_, { n }: { n: number }) => {
    const mod = getNative()
    if (!mod) return null

    const jsStart = performance.now()
    let a = 0n, b = 1n
    for (let i = 0; i < n; i++) [a, b] = [b, a + b]
    const jsDuration = performance.now() - jsStart

    const rustStart = performance.now()
    const rustResult = mod.benchmarkFib(n)
    const rustDuration = performance.now() - rustStart

    return {
      js: { result: String(a), durationMs: jsDuration },
      rust: { result: String(rustResult), durationMs: rustDuration },
      speedup: jsDuration / rustDuration
    }
  })
}
```

## 5. Build Script

```bash
#!/bin/bash
# scripts/build-native.sh

set -e

cd native

# สร้าง Rust native module
echo "Building Rust native module..."
cargo build --release

# Generate TypeScript bindings
echo "Generating TypeScript bindings..."
napi build --platform --release

echo "Native module built successfully!"

# Copy .node file ไปยัง resources
mkdir -p ../resources/native
cp *.node ../resources/native/ 2>/dev/null || true
```

```json
// electron-builder.json5 (เพิ่ม native module)
{
  "asarUnpack": [

  ],
  "extraFiles": [
    {
      "from": "native",
      "to": "resources/native",
      "filter": ["*.node"]
    }
  ]
}
```

## สรุป

| ฟีเจอร์ | NAPI-RS | ประสิทธิภาพ |
|--------|---------|------------|
| Sync functions | #[napi] fn | Fast, blocks JS thread |
| Struct/class | #[napi] impl | มี state |
| Async functions | async fn + tokio | ไม่ block JS |
| Parallel processing | rayon | CPU multi-core |
| Memory safety | Rust borrow checker | ไม่มี segfault |
| Cross-compilation | napi targets | Win/Mac/Linux |
| Auto-generated types | .d.ts | Type-safe JS |
